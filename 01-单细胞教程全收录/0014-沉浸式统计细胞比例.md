cols <-pal_npg("nrc")(10)#cols##  [1] "#E64B35FF" "#4DBBD5FF" "#00A087FF" "#3C5488FF" "#F39B7FFF" "#8491B4FF"##  [7] "#91D1C2FF" "#DC0000FF" "#7E6148FF" "#B09C85FF"
