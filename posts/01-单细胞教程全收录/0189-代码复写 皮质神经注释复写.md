#####安装包install.packages("RCurl")install.packages("BiocManager")BiocManager::install("SingleCellExperiment")BiocManager::install("tidyverse")BiocManager::install('multtest')install.packages('Seurat')install.packages('hdf5r')install.packages("dplyr")install.packages("ggplot2")install.packages("stringr")#####加载包library(SingleCellExperiment)library(Seurat)library(tidyverse)library(Matrix)library(scales)library(cowplot)library(RCurl)library(dplyr)library(ggplot2)library(stringr)library(metap)library(Seurat)library(patchwork)

####————————————————————————————————————————————————————————————————————导入数据建立Seurat对象"SYNGAP1_1"for (file in c("patient_1","patient_2", "patient_3","SYNGAP1_1","SYNGAP1_2","SYNGAP1_3")) {  seurat_data <- Read10X(data.dir = paste0("/home/zhangxu/re/2m_", file))  seurat_obj <- CreateSeuratObject(counts = seurat_data,                                   min.cells = 3, #某一个基因至少在多少个细胞中表达                                   min.features = 200,#某个细胞至少表达多少个基因                                   project = file)  assign(file, seurat_obj)}
# ### 创建一个合并的Seurat对象merged_seurat <- merge(x = patient_1,                       y = c(patient_2,patient_3,SYNGAP1_1,SYNGAP1_2,SYNGAP1_3),                        add.cell.id = c("P1","P2","P3","S1","S2","S3"))
####————————————————————————————————————————————————————————————————————生成质量指标# 将每个细胞每个UMI的基因数目添加到元数据中merged_seurat$log10GenesPerUMI <- log10(merged_seurat$nFeature_RNA) / log10(merged_seurat$nCount_RNA)
# 计算线粒体比率merged_seurat$mitoRatio <- PercentageFeatureSet(object = merged_seurat, pattern = "^MT-")merged_seurat$mitoRatio <- merged_seurat@meta.data$mitoRatio / 100
# 创建元数据数据框metadata <- merged_seurat@meta.data# 为元数据添加细胞IDmetadata$cells <- rownames(metadata)# 重命名列metadata <- metadata %>%  dplyr::rename(seq_folder = orig.ident,                nUMI = nCount_RNA,                nGene = nFeature_RNA)# 创建样本列metadata$sample <- NAmetadata$sample[which(str_detect(metadata$cells, "^P1"))] <- "P1"metadata$sample[which(str_detect(metadata$cells, "^P2"))] <- "P2"metadata$sample[which(str_detect(metadata$cells, "^P3"))] <- "P3"metadata$sample[which(str_detect(metadata$cells, "^S1"))] <- "S1"metadata$sample[which(str_detect(metadata$cells, "^S2"))] <- "S2"metadata$sample[which(str_detect(metadata$cells, "^S3"))] <- "S3"
# 将元数据添加回Seurat对象中merged_seurat@meta.data <- metadata
#####————————————————————————————————————————————————————————————————————————筛选细胞# 使用选择的阈值筛掉低质量读写 - 这些将会随实验而改变filtered_seurat <- subset(x = merged_seurat,                           subset= (nUMI >= 1250) &                             (mitoRatio < 0.15))  table(filtered_seurat[[]]$sample)# ####_____________________________________________________________________细胞周期评分# 导入细胞周期标记物load("data/cycle.rda")
# 给细胞的细胞周期评分seurat_phase <- CellCycleScoring(filtered_seurat,                                 g2m.features = g2m_genes,                                 s.features = s_genes)table(seurat_phase[[]]$Phase)# ####_____________________________________________________________________SCT、PCA、UMAP、聚类
seurat_phase <- SCTransform(seurat_phase,                            vars.to.regress = c("mitoRatio","S.Score","G2M.Score"))
seurat_phase <- RunPCA(seurat_phase)ElbowPlot(seurat_phase,ndims = 40)seurat_phase <- FindNeighbors(seurat_phase, dims = 1:30)seurat_phase <- FindClusters(seurat_phase,resolution = c(0.2, 0.8, 1.2)) seurat_phase <- RunUMAP(seurat_phase, dims = 1:30)
DimPlot(seurat_phase)
# 保存聚合的R对象saveRDS(seurat_phase, "/home/re/seurat_phase6.rds")
#————————————————————————————————————————————————————————————————————聚类分析
# 探索分辨率seurat_phase@meta.data %>%  View()# 查看特异基因表达DotPlot(seurat_phase,features = c("PAX6", "SOX2", "HOPX", "EOMES", "NEUROD1", "TBR1", "GAD1", "GAD2", "SLC32A1", "BCL11B","CUX1", "POU3F2", "EMX1"))
FeaturePlot(seurat_phase,            reduction = "umap",            features = c("PAX6", "SOX2", "HOPX", "EOMES", "NEUROD1",                          "TBR1", "GAD1", "GAD2", "SLC32A1", "BCL11B",                         "CUX1", "POU3F2", "EMX1"),            order = TRUE,            min.cutoff = 'q10',            label =FALSE)
# 提取差异基因all_markers <- FindAllMarkers(seurat_phase                              , min.pct = 0.25, logfc.threshold = 0.25)top25 <- all_markers %>% group_by(cluster) %>% top_n(-25,p_val_adj)
# 赋予聚类的身份Idents(object = seurat_phase) <- "SCT_snn_res.1.2"seurat_phase <- RenameIdents(object = seurat_phase,                         "0" = "aRG",                        "11" = "aRG",                        "18" = "aRG",                                                "7" = "oRG",                        "9" = "oRG",                        "12" = "oRG",                        "14" = "oRG",                        "16" = "oRG",                        "22" = "oRG",                        "25" = "oRG",                                                "17" = "IP",                        "20" = "IP",                        "21" = "IP",                        "10" = "IP",                        "26" = "IP",                                                                        "1" = "Unspecified PN",                        "2" = "Unspecified PN",                        "3" = "Unspecified PN",                        "4" = "Unspecified PN",                        "5" = "Unspecified PN",                        "13" = "Unspecified PN",                        "19" = "Unspecified PN",                                                "1" = "Mature neuron",                        "4" = "Mature neuron",                        "12" = "Mature neuron",                        "16" = "Mature neuron",
                        "15" = "Inhibitory N",                                                "6" = "Newborn CPN",                                                "8" ="Newborn CFuPN",                                                "23" ="CFuPN",                                                "24" = "Cortical hem")

DimPlot(seurat_phase,         label = TRUE,         label.size = 3,        split.by = NULL)######按照组别分离类群DimPlot(seurat_phase,         label = TRUE,         label.size = 3,        split.by = "sample")
seurat_phase$celltypes <- Idents(seurat_phase)
# 保存最终的R对象write_rds(seurat_phase,file = "results/seurat_labelled.rds")
